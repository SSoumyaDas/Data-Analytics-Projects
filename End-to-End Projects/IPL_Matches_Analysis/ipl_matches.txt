CREATE TABLE ipl_matches (
    id NUMBER,
    season VARCHAR2(10),
    city VARCHAR2(100),
    match_date DATE,
    match_type VARCHAR2(50),
    player_of_match VARCHAR2(100),
    venue VARCHAR2(200),
    team1 VARCHAR2(100),
    team2 VARCHAR2(100),
    toss_winner VARCHAR2(100),
    toss_decision VARCHAR2(10),
    winner VARCHAR2(100),
    result VARCHAR2(20),
    result_margin NUMBER,
    target_runs NUMBER,
    target_overs NUMBER,
    super_over VARCHAR2(5),
    method VARCHAR2(20),
    umpire1 VARCHAR2(100),
    umpire2 VARCHAR2(100),
    toss_winner_won VARCHAR2(5)
);

Q4. Which teams have the highest win percentage?

SELECT team,
       COUNT(*) AS matches_played,
       SUM(CASE WHEN winner = team THEN 1 ELSE 0 END) AS total_wins,
       ROUND(SUM(CASE WHEN winner = team THEN 1 ELSE 0 END) / COUNT(*) * 100, 2) AS win_percentage
FROM (
    SELECT team1 AS team, winner FROM ipl_matches
    UNION ALL
    SELECT team2 AS team, winner FROM ipl_matches
)
GROUP BY team
ORDER BY win_percentage DESC;

Q5. What is the toss decision distribution?

SELECT toss_decision,
       COUNT(*) AS total_matches,
       ROUND(COUNT(*) / (SELECT COUNT(*) FROM ipl_matches) * 100, 2) AS percentage
FROM ipl_matches
GROUP BY toss_decision
ORDER BY total_matches DESC;

Q8.Which players have won the most Player of the Match awards?

SELECT player_of_match,
       awards
FROM (
    SELECT player_of_match,
           COUNT(*) AS awards
    FROM ipl_matches
    WHERE player_of_match IS NOT NULL
    GROUP BY player_of_match

WHERE ROWNUM <= 5;
