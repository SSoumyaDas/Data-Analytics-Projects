CREATE TABLE customer_behavior (
    customer_id NUMBER,
    age NUMBER,
    gender VARCHAR2(10),
    item_purchased VARCHAR2(50),
    category VARCHAR2(30),
    purchase_amount NUMBER(10,2),
    location VARCHAR2(30),
    product_size VARCHAR2(5),
    color VARCHAR2(30),
    season VARCHAR2(15),
    review_rating NUMBER(3,1),
    subscription_status VARCHAR2(5),
    shipping_type VARCHAR2(30),
    discount_applied VARCHAR2(5),
    previous_purchases NUMBER,
    payment_method VARCHAR2(30),
    frequency_of_purchases VARCHAR2(30),
    age_group VARCHAR2(20),
    purchase_frequency_days NUMBER
);

Q19.How many customers fall into each spending category?
    High Spender → purchase amount ≥ 80
    Medium Spender → 50–79
    Low Spender → below 50

SELECT
    CASE
        WHEN purchase_amount >= 80 THEN 'High Spender'
        WHEN purchase_amount >= 50 THEN 'Medium Spender'
        ELSE 'Low Spender'
    END AS spending_category,
    COUNT(*) AS customer_count
FROM customer_behavior
GROUP BY
    CASE
        WHEN purchase_amount >= 80 THEN 'High Spender'
        WHEN purchase_amount >= 50 THEN 'Medium Spender'
        ELSE 'Low Spender'
    END
ORDER BY customer_count DESC;


SPENDING_CATEG CUSTOMER_COUNT
-------------- --------------
Low Spender              1466
Medium Spender           1410
High Spender             1024
