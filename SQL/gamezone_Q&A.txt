🟢 LEVEL 1 — SIMPLE

-- Q1.Find the total number of players registered on the platform.

SELECT COUNT(*) AS total_players
FROM players;

-- Q2.Display each game along with its genre, sorted alphabetically by game name.

SELECT
    game_name,
    genre
FROM games
ORDER BY game_name;

-- Q3.Find the number of players from each country. Display only countries having more than one player.

SELECT
    country,
    COUNT(*) AS total_players
FROM players
GROUP BY country
HAVING COUNT(*) > 1
ORDER BY total_players DESC;

-- Q4.Calculate the total number of gaming sessions and the average session duration.

SELECT
    COUNT(*) AS total_sessions,
    ROUND(AVG(duration_minutes), 2) AS avg_session_duration
FROM sessions;

-- Q5.Find the total number of sessions played for each game.

SELECT
    g.game_name,
    COUNT(s.session_id) AS total_sessions
FROM games g
JOIN sessions s
    ON g.game_id = s.game_id
GROUP BY g.game_name
ORDER BY total_sessions DESC;

🟡 LEVEL 2 — MEDIUM

-- Q6.Calculate the total revenue generated by each game.

SELECT
    g.game_name,
    SUM(p.amount) AS total_revenue
FROM games g
JOIN purchases p
    ON g.game_id = p.game_id
GROUP BY g.game_name
ORDER BY total_revenue DESC;

-- Q7.Find the total amount spent by each player.

SELECT
    pl.player_name,
    SUM(p.amount) AS total_spending
FROM players pl
JOIN purchases p
    ON pl.player_id = p.player_id
GROUP BY pl.player_name
ORDER BY total_spending DESC;

-- Q8.Find the top 5 players based on their total gaming time.

SELECT *
FROM (
    SELECT
        pl.player_name,
        COUNT(s.session_id) AS total_sessions,
        SUM(s.duration_minutes) AS total_gaming_minutes
    FROM players pl
    JOIN sessions s
        ON pl.player_id = s.player_id
    GROUP BY pl.player_name
    ORDER BY total_gaming_minutes DESC
)
WHERE ROWNUM <= 5;


-- Q9.For each genre, calculate the number of games, total sessions, and total gaming minutes.

SELECT
    g.genre,
    COUNT(DISTINCT g.game_id) AS total_games,
    COUNT(s.session_id) AS total_sessions,
    SUM(s.duration_minutes) AS total_gaming_minutes
FROM games g
JOIN sessions s
    ON g.game_id = s.game_id
GROUP BY g.genre
ORDER BY total_gaming_minutes DESC;


-- Q10.Find players whose total spending is greater than the average spending of all players who have made a purchase.


SELECT
    pl.player_name,
    SUM(p.amount) AS total_spending
FROM players pl
JOIN purchases p
    ON pl.player_id = p.player_id
GROUP BY pl.player_name
HAVING SUM(p.amount) > (
    SELECT AVG(total_spending)
    FROM (
        SELECT
            player_id,
            SUM(amount) AS total_spending
        FROM purchases
        GROUP BY player_id
    )
)
ORDER BY total_spending DESC;


🔴 LEVEL 3 — ADVANCED

-- Q11.Rank games by total revenue within each genre.

SELECT
    genre,
    game_name,
    total_revenue,
    RANK() OVER (
        PARTITION BY genre
        ORDER BY total_revenue DESC
    ) AS revenue_rank
FROM (
    SELECT
        g.genre,
        g.game_name,
        SUM(p.amount) AS total_revenue
    FROM games g
    JOIN purchases p
        ON g.game_id = p.game_id
    GROUP BY
        g.genre,
        g.game_name
)
ORDER BY genre, revenue_rank;


-- Q12.Rank all players according to their total gaming time.

SELECT
    player_name,
    total_gaming_minutes,
    RANK() OVER (
        ORDER BY total_gaming_minutes DESC
    ) AS player_rank
FROM (
    SELECT
        pl.player_name,
        SUM(s.duration_minutes) AS total_gaming_minutes
    FROM players pl
    JOIN sessions s
        ON pl.player_id = s.player_id
    GROUP BY pl.player_name
)
ORDER BY player_rank;


-- Q13. Calculate cumulative revenue over time.


SELECT
    purchase_date,
    daily_revenue,
    SUM(daily_revenue) OVER (
        ORDER BY purchase_date
        ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS cumulative_revenue
FROM (
    SELECT
        purchase_date,
        SUM(amount) AS daily_revenue
    FROM purchases
    GROUP BY purchase_date
)
ORDER BY purchase_date;



-- Q14.For each player, identify the game they spent the most time playing.

SELECT
    player_name,
    game_name,
    total_gaming_minutes
FROM (
    SELECT
        pl.player_name,
        g.game_name,
        SUM(s.duration_minutes) AS total_gaming_minutes,
        ROW_NUMBER() OVER (
            PARTITION BY pl.player_id
            ORDER BY SUM(s.duration_minutes) DESC
        ) AS game_rank
    FROM players pl
    JOIN sessions s
        ON pl.player_id = s.player_id
    JOIN games g
        ON s.game_id = g.game_id
    GROUP BY
        pl.player_id,
        pl.player_name,
        g.game_name
)
WHERE game_rank = 1
ORDER BY player_name;

-- Q15.Create a player-level analysis containing total sessions, total gaming minutes, total spending, average session duration, and spending rank.


SELECT
    player_name,
    total_sessions,
    total_gaming_minutes,
    total_spending,
    avg_session_duration,
    RANK() OVER (
        ORDER BY total_spending DESC
    ) AS spending_rank
FROM (
    SELECT
        pl.player_id,
        pl.player_name,
        COUNT(DISTINCT s.session_id) AS total_sessions,
        NVL(SUM(s.duration_minutes), 0) AS total_gaming_minutes,
        NVL(SUM(p.amount), 0) AS total_spending,
        ROUND(AVG(s.duration_minutes), 2) AS avg_session_duration
    FROM players pl
    LEFT JOIN sessions s
        ON pl.player_id = s.player_id
    LEFT JOIN purchases p
        ON pl.player_id = p.player_id
    GROUP BY
        pl.player_id,
        pl.player_name
)
ORDER BY spending_rank;
